    <!-- ==================================================

         ================================================== -->

    <script>


        // =================================================
        // CONFIGURACIÓN DEL MODELO
        // =================================================


        /*
         * IMPORTANTE:
         *
         * REEMPLAZAR ESTA URL POR LA URL
         * DEL MODELO CREADO EN TEACHABLE MACHINE.
         *
         * Ejemplo:
         *
         * const URL =
         * "https://teachablemachine.withgoogle.com/models/ABC123/";
         *
         */


        const URL =
        "https://teachablemachine.withgoogle.com/models/XXXXXXXX/";


        // =================================================
        // VARIABLES GLOBALES
        // =================================================


        let model;

        let webcam;

        let maxPredictions;

        let ejecutando = false;


        // =================================================
        // INICIAR CÁMARA Y MODELO
        // =================================================


        async function iniciar() {


            try {


                // Cambiar estado

                document.getElementById("estado")
                    .innerHTML =
                    "⏳ Cargando modelo...";


                // Crear direcciones del modelo

                const modelURL =
                    URL + "model.json";


                const metadataURL =
                    URL + "metadata.json";


                // Cargar modelo

                model = await tmImage.load(
                    modelURL,
                    metadataURL
                );


                // Obtener número de clases

                maxPredictions =
                    model.getTotalClasses();


                // Crear cámara

                const flip = true;


                webcam =
                    new tmImage.Webcam(
                        400,
                        400,
                        flip
                    );


                // Configurar cámara

                await webcam.setup();


                // Encender cámara

                await webcam.play();


                // Activar ejecución

                ejecutando = true;


                // Ocultar resultado anterior

                document.getElementById(
                    "resultado-principal"
                ).style.display = "none";


                // Limpiar cámara anterior

                document.getElementById(
                    "webcam-container"
                ).innerHTML = "";


                // Mostrar cámara

                document.getElementById(
                    "webcam-container"
                ).appendChild(
                    webcam.canvas
                );


                // Crear contenedores
                // para las predicciones

                const labelContainer =
                    document.getElementById(
                        "label-container"
                    );


                labelContainer.innerHTML = "";


                for (
                    let i = 0;
                    i < maxPredictions;
                    i++
                ) {


                    const resultado =
                        document.createElement(
                            "div"
                        );


                    resultado.className =
                        "resultado";


                    resultado.innerHTML = `

                        <div class="nombre-clase">

                            Clase ${i + 1}

                        </div>

                        <div class="barra-fondo">

                            <div
                                id="barra-${i}"
                                class="barra">
                            </div>

                        </div>

                        <div
                            id="porcentaje-${i}"
                            class="porcentaje">

                            0%

                        </div>

                    `;


                    labelContainer.appendChild(
                        resultado
                    );

                }


                // Cambiar botones

                document.getElementById(
                    "btnIniciar"
                ).disabled = true;


                document.getElementById(
                    "btnDetener"
                ).disabled = false;


                // Cambiar estado

                document.getElementById(
                    "estado"
                ).innerHTML =
                    "🟢 Cámara activa - Analizando...";


                // Comenzar ciclo

                window.requestAnimationFrame(
                    loop
                );


            } catch (error) {


                console.error(error);


                document.getElementById(
                    "estado"
                ).innerHTML =
                    "❌ Error al iniciar la cámara o cargar el modelo.";


                alert(
                    "No se pudo iniciar el modelo o la cámara.\n\n" +
                    "Verifique la URL del modelo y los permisos de la cámara."
                );

            }

        }


        // =================================================
        // BUCLE DE PREDICCIÓN
        // =================================================


        async function loop() {


            if (!ejecutando) {

                return;

            }


            // Actualizar cámara

            webcam.update();


            // Realizar predicción

            await predecir();


            // Continuar

            window.requestAnimationFrame(
                loop
            );

        }


        // =================================================
        // REALIZAR PREDICCIÓN
        // =================================================


        async function predecir() {


            if (!model || !webcam) {

                return;

            }


            // Obtener predicciones

            const prediction =
                await model.predict(
                    webcam.canvas
                );


            // Variables para encontrar
            // la clase con mayor probabilidad

            let mayorProbabilidad = 0;

            let clasePrincipal = "";


            // Recorrer todas las clases

            for (
                let i = 0;
                i < maxPredictions;
                i++
            ) {


                // Nombre de la clase

                const nombre =
                    prediction[i].className;


                // Probabilidad

                const probabilidad =
                    prediction[i].probability;


                // Convertir a porcentaje

                const porcentaje =
                    (
                        probabilidad * 100
                    ).toFixed(1);


                // Actualizar nombre

                const nombreElement =
                    document.querySelector(
                        `#label-container .resultado:nth-child(${i + 1}) .nombre-clase`
                    );


                if (nombreElement) {

                    nombreElement.innerHTML =
                        nombre;

                }


                // Actualizar barra

                const barra =
                    document.getElementById(
                        `barra-${i}`
                    );


                if (barra) {

                    barra.style.width =
                        porcentaje + "%";

                }


                // Actualizar porcentaje

                const porcentajeElement =
                    document.getElementById(
                        `porcentaje-${i}`
                    );


                if (porcentajeElement) {

                    porcentajeElement.innerHTML =
                        porcentaje + "%";

                }


                // Buscar mayor probabilidad

                if (
                    probabilidad >
                    mayorProbabilidad
                ) {

                    mayorProbabilidad =
                        probabilidad;

                    clasePrincipal =
                        nombre;

                }

            }


            // Mostrar resultado principal

            const resultadoPrincipal =
                document.getElementById(
                    "resultado-principal"
                );


            resultadoPrincipal.style.display =
                "block";


            document.getElementById(
                "prediccion-principal"
            ).innerHTML =

                clasePrincipal +
                "<br>" +
                "<small>" +
                (
                    mayorProbabilidad * 100
                ).toFixed(1) +
                "% de confianza" +
                "</small>";

        }


        // =================================================
        // DETENER CÁMARA
        // =================================================


        function detener() {


            ejecutando = false;


            // Detener cámara

            if (webcam) {

                webcam.stop();

            }


            // Eliminar imagen de cámara

            document.getElementById(
                "webcam-container"
            ).innerHTML = "";


            // Limpiar resultados

            document.getElementById(
                "label-container"
            ).innerHTML = "";


            document.getElementById(
                "resultado-principal"
            ).style.display = "none";


            // Actualizar estado

            document.getElementById(
                "estado"
            ).innerHTML =
                "🔴 Cámara detenida";


            // Activar botón iniciar

            document.getElementById(
                "btnIniciar"
            ).disabled = false;


            // Desactivar botón detener

            document.getElementById(
                "btnDetener"
            ).disabled = true;

        }


    </script>
